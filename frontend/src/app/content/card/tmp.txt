  <script>
    const card = document.getElementById('card');
    const badge = document.getElementById('badge');
    const sel = document.getElementById('school');
    const exportBtn = document.getElementById('exportBtn');

    const schools = {
      evocation: {label:'Ev', class:'school-evocation', color:'#d94a4a'},
      conjuration: {label:'Cj', class:'school-conjuration', color:'#2b8fb7'},
      abjuration: {label:'Ab', class:'school-abjuration', color:'#3b7a3d'},
      transmutation: {label:'Tr', class:'school-transmutation', color:'#c77b2b'},
      necromancy: {label:'Nc', class:'school-necromancy', color:'#5b3b83'},
      illusion: {label:'Il', class:'school-illusion', color:'#7b5aa2'},
      enchantment: {label:'En', class:'school-enchantment', color:'#b24f9b'},
      divination: {label:'Dv', class:'school-divination', color:'#3b6fa8'}
    };

    function applySchool(s){
      // remove previous school- classes
      Object.values(schools).forEach(sch=>card.classList.remove(sch.class));
      card.classList.add(schools[s].class);
      badge.textContent = schools[s].label;
      // set CSS variable for custom color fallback
      card.style.setProperty('--school-color', schools[s].color);
    }

    sel.addEventListener('change', e=>applySchool(e.target.value));

    exportBtn.addEventListener('click', ()=>{
      window.print();
    });

    // UX: make contenteditable fields behave nicely
    document.querySelectorAll('[contenteditable]').forEach(el=>{
      el.addEventListener('focus', ()=>el.dataset.old = el.innerText);
      el.addEventListener('blur', ()=>{
        if(!el.innerText.trim()) el.innerText = el.dataset.old || '';
      });
    });

    // initialize
    applySchool(sel.value);
  </script>
